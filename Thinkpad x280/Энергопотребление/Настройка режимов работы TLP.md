### Шаг 1: Создаем/открываем файл `.bash_aliases`
```bash
nano ~/.bash_aliases_tlp
```
### Шаг 2: Копируем и вставляем весь блок
```bash
# ==============================================================================
# АЛИАСЫ ДЛЯ УПРАВЛЕНИЯ TLP (ThinkPad X280)
# ==============================================================================

# ------------------------------------------------------------------------------
# РЕЖИМЫ ПРОИЗВОДИТЕЛЬНОСТИ ПРОЦЕССОРА
# ------------------------------------------------------------------------------

# РЕЖИМ 1: МАКСИМАЛЬНАЯ ПРОИЗВОДИТЕЛЬНОСТЬ (TURBO)
# Для чего: Игры, рендеринг, тяжёлые задачи (даже от батареи)
# Что делает: Turbo ON | Performance | Заряд до 100% | USB без экономии
alias tlp-turbo='echo 0 | sudo tee /sys/devices/system/cpu/intel_pstate/no_turbo; echo performance | sudo tee /sys/devices/system/cpu/cpu*/cpufreq/scaling_governor; sudo tlp bat none; sudo tlp usb on; echo "Турбо режим: ON | Батарея: заряд до 100% | USB: без энергосбережения"'

# РЕЖИМ 2: СБАЛАНСИРОВАННЫЙ (AUTO)
# Для чего: Обычная работа, стандартное поведение TLP
# Что делает: Turbo как решит TLP | Perf от AC / Powersave от BAT | Батарея 60-80%
alias tlp-auto='sudo systemctl restart tlp; sudo tlp bat 60 80; sudo tlp usb auto; echo "Авто режим: стандартное поведение TLP | Батарея: 60-80%"'

# РЕЖИМ 3: ЭКОНОМИЯ (BATTERY SAVER)
# Для чего: Максимальное время автономной работы
# Что делает: Turbo OFF | Powersave | Лимит 2.1 GHz | Батарея 60-80%
alias tlp-battery='sudo systemctl stop tlp; echo 1 | sudo tee /sys/devices/system/cpu/intel_pstate/no_turbo; echo powersave | sudo tee /sys/devices/system/cpu/cpu*/cpufreq/scaling_governor; sudo cpupower frequency-set -u 2100000 2>/dev/null; sudo tlp start; sudo tlp bat 60 80; sudo tlp usb auto; echo "Эконом режим: Turbo OFF | Частота лимит: 2.1 GHz | Батарея: 60-80%"'

# РЕЖИМ 4: МАКСИМАЛЬНАЯ ЭКОНОМИЯ (MAX SAVE)
# Для чего: Когда нужно "выжать" последние минуты
# Что делает: Turbo OFF | 0.8-1.5 GHz | Батарея 60-80%
alias tlp-max-save='echo 1 | sudo tee /sys/devices/system/cpu/intel_pstate/no_turbo; echo powersave | sudo tee /sys/devices/system/cpu/cpu*/cpufreq/scaling_governor; sudo tlp bat 60 80; sudo cpupower frequency-set -u 1500000 -d 800000 2>/dev/null; sudo tlp usb auto; echo "MAX SAVE: частота 0.8-1.5 GHz | Батарея: 60-80%"'

# СБРОС НАСТРОЕК
# Что делает: Возвращает к конфигурации по умолчанию
alias tlp-reset='sudo systemctl restart tlp && echo "Настройки сброшены к конфигу /etc/tlp.conf"'

# ------------------------------------------------------------------------------
# РЕЖИМЫ УПРАВЛЕНИЯ БАТАРЕЕЙ
# ------------------------------------------------------------------------------

# РЕЖИМ 5: ПОЛНАЯ ЗАРЯДКА (В ДОРОГУ)
# Для чего: Когда знаете, что ближайшие часы не будет розетки
# Что делает: Отключает пороги и заряжает батарею до 100%
alias bat-max='sudo tlp fullcharge && echo "Режим: ПОЛНАЯ ЗАРЯДКА (до 100%)"'

# РЕЖИМ 6: НОРМАЛЬНАЯ ЗАРЯДКА (60-80%)
# Для чего: Повседневное использование, продление жизни батареи
# Что делает: Устанавливает пороги 60-80%
alias bat-normal='sudo tlp setcharge 60 80 && echo "Режим: НОРМАЛЬНЫЙ (60-80%)"'

# ------------------------------------------------------------------------------
# ИНФОРМАЦИОННЫЕ АЛИАСЫ
# ------------------------------------------------------------------------------

# СТАТУС БАТАРЕИ
# Показывает: текущий заряд и пороги
alias bat-status='sudo tlp-stat -b | grep -E "charge|threshold"'

# ПОЛНЫЙ СТАТУС TLP
# Показывает: режим питания и состояние батареи
alias tlp-status='sudo tlp-stat -s && sudo tlp-stat -b'

# ------------------------------------------------------------------------------
# КОНЕЦ НАСТРОЕК TLP
# ------------------------------------------------------------------------------
# ==============================================================================
# ФУНКЦИЯ АВТОМАТИЧЕСКОГО ПАРСИНГА АЛИАСОВ TLP
# ==============================================================================
alias-tlp() {
    local file="$HOME/.bash_aliases_tlp"  # ← фиксированный путь к файлу
    
    echo ""
    echo -e "\033[1;36m════════════════════════════════════════════════════════════════════════════════\033[0m"
    echo -e "\033[1;33m                       УПРАВЛЕНИЕ TLP — ThinkPad X280\033[0m"
    echo -e "\033[1;36m════════════════════════════════════════════════════════════════════════════════\033[0m"
    echo ""
    
    local section=""
    local description=""
    local purpose=""
    local action=""
    local alias_name=""
    
    while IFS= read -r line; do
        # Определяем секцию
        if [[ "$line" =~ ^#[[:space:]]-+[[:space:]]*РЕЖИМЫ[[:space:]]ПРОИЗВОДИТЕЛЬНОСТИ ]]; then
            section="perf"
            echo -e "\033[1;32m▶ РЕЖИМЫ ПРОИЗВОДИТЕЛЬНОСТИ ПРОЦЕССОРА\033[0m"
            echo -e "\033[36m────────────────────────────────────────────────────────────────────────────────\033[0m"
        elif [[ "$line" =~ ^#[[:space:]]-+[[:space:]]*РЕЖИМЫ[[:space:]]УПРАВЛЕНИЯ[[:space:]]БАТАРЕЕЙ ]]; then
            section="bat"
            echo -e "\033[1;32m▶ РЕЖИМЫ УПРАВЛЕНИЯ БАТАРЕЕЙ\033[0m"
            echo -e "\033[36m────────────────────────────────────────────────────────────────────────────────\033[0m"
        elif [[ "$line" =~ ^#[[:space:]]-+[[:space:]]*ИНФОРМАЦИОННЫЕ ]]; then
            section="info"
            echo -e "\033[1;32m▶ ИНФОРМАЦИОННЫЕ КОМАНДЫ\033[0m"
            echo -e "\033[36m────────────────────────────────────────────────────────────────────────────────\033[0m"
        fi
        
        # Собираем описание
        if [[ "$line" =~ ^#[[:space:]]*РЕЖИМ[[:space:]][0-9]+:[[:space:]](.+) ]]; then
            description="${BASH_REMATCH[1]}"
        elif [[ "$line" =~ ^#[[:space:]]*Для[[:space:]]чего:[[:space:]](.+) ]]; then
            purpose="${BASH_REMATCH[1]}"
        elif [[ "$line" =~ ^#[[:space:]]*Что[[:space:]]делает:[[:space:]](.+) ]]; then
            action="${BASH_REMATCH[1]}"
        elif [[ "$line" =~ ^#[[:space:]]*Показывает:[[:space:]](.+) ]]; then
            action="Показывает: ${BASH_REMATCH[1]}"
            purpose=""
        fi
        
        # Выводим алиас
        if [[ "$line" =~ ^alias[[:space:]]+([^=]+)= ]]; then
            alias_name="${BASH_REMATCH[1]}"
            
            echo -e " \033[1;35m${alias_name}\033[0m"
            
            if [[ -n "$description" ]]; then
                echo -e "                  \033[1;37m${description}\033[0m"
            fi
            
            if [[ -n "$purpose" ]]; then
                echo -e "                  ${purpose}"
            fi
            
            if [[ -n "$action" ]]; then
                echo -e "                  \033[90m→ ${action}\033[0m"
            fi
            
            echo ""
            
            # Сбрасываем переменные
            description=""
            purpose=""
            action=""
        fi
    done < "$file"
    
    echo -e "\033[1;36m════════════════════════════════════════════════════════════════════════════════\033[0m"
    echo -e "\033[1;33mИспользуйте \033[1;37malias-tlp\033[1;33m для повторного вызова этого меню\033[0m"
    echo -e "\033[1;36m════════════════════════════════════════════════════════════════════════════════\033[0m"
    echo ""
}
```
### Шаг 3: Сохраняем файл
- Нажмите `Ctrl+O` (сохранить)
- Нажмите `Enter` (подтвердить имя файла)
- Нажмите `Ctrl+X` (выйти)
### Шаг 4: Подключаем в `.bashrc` алиасы `.bash_aliases_tlp`

```bash
nano ~/.bashrc
```

Проверьте, есть ли в файле строки (обычно они есть в стандартном `.bashrc`):
```bash
if [ -f ~/.bash_aliases ]; then
    . ~/.bash_aliases
fi

# Добавить эти строки
if [ -f ~/.bash_aliases_tlp ]; then
    source ~/.bash_aliases_tlp
fi
```
### Шаг 5: Применяем изменения

```bash
source ~/.bashrc
```
### Шаг 7: Настройка `sudo` без пароля (опционально, но удобно)

Чтобы не вводить пароль для каждой команды:
```bash
sudo visudo
```

Добавьте строку (замените `username` на ваше имя пользователя):
```text
username ALL=(ALL) NOPASSWD: /usr/bin/tlp, /usr/bin/tee, /usr/bin/cpupower, /usr/bin/tlp-stat
```
## ТАБЛИЦА РЕЖИМОВ TLP

| Режим            | Turbo Boost            | Губернатор                | Частота CPU               | Заряд батареи    | USB          | Для чего нужен                                                                                       |
| ---------------- | ---------------------- | ------------------------- | ------------------------- | ---------------- | ------------ | ---------------------------------------------------------------------------------------------------- |
| **tlp-turbo**    | ✅ ВКЛ                  | performance               | макс (до 3.4 GHz)         | до 100% (полная) | без экономии | **Игры, рендеринг, тяжёлые задачи**. Когда нужна максимальная мощность, даже если ноутбук на батарее |
| **tlp-auto**     | 🔄 AC: ВКЛ / BAT: ВЫКЛ | AC: perf / BAT: powersave | без лимита                | 60-80% (защита)  | авто         | **Повседневная работа**. Умное переключение: от сети — быстро, от батареи — экономно                 |
| **tlp-battery**  | ❌ ВЫКЛ                 | powersave                 | до 2.1 GHz                | 60-80% (защита)  | авто         | **Долгая работа от батареи**. Когда нужно проработать несколько часов без розетки                    |
| **tlp-max-save** | ❌ ВЫКЛ                 | powersave                 | **0.8-1.5 GHz** (минимум) | 60-80% (защита)  | авто         | **"Выжать" последние минуты**. Когда батарея на исходе, а розетки нет                                |
